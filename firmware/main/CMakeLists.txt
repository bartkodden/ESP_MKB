cmake_minimum_required(VERSION 3.16)

set(COMPONENT_SRCDIR ${CMAKE_CURRENT_SOURCE_DIR})

# ─── EEZ / UI sources ────────────────────────────────────────────────────────
file(GLOB UI_C_SOURCES   "${COMPONENT_SRCDIR}/ui/*.c")
file(GLOB UI_CPP_SOURCES "${COMPONENT_SRCDIR}/ui/*.cpp")

# ─── Firmware-embedded static assets (img/ only — small, never-changing) ─────
file(GLOB EMBEDDED_IMAGES "${COMPONENT_SRCDIR}/img/*.png")

# ─── Python ──────────────────────────────────────────────────────────────────
find_package(Python3 REQUIRED COMPONENTS Interpreter)

# ─── menu.json generation ────────────────────────────────────────────────────
# Generated from menu.json_example + .env (local) or defaults (CI).
# If menu.json already exists (pre-generated by CI step), skip generation.
set(GEN_MENU_SCRIPT  "${COMPONENT_SRCDIR}/scripts/gen_menu.py")
set(MENU_EXAMPLE     "${COMPONENT_SRCDIR}/menu.json_example")
set(MENU_ENV         "${COMPONENT_SRCDIR}/.env")
set(MENU_OUT         "${COMPONENT_SRCDIR}/menu.json")

if(NOT CMAKE_SCRIPT_MODE_FILE)
    if(NOT EXISTS "${MENU_OUT}")
        add_custom_command(
            OUTPUT  ${MENU_OUT}
            COMMAND ${Python3_EXECUTABLE} ${GEN_MENU_SCRIPT}
            DEPENDS ${GEN_MENU_SCRIPT} ${MENU_EXAMPLE} ${MENU_ENV}
            WORKING_DIRECTORY ${COMPONENT_SRCDIR}/scripts
            COMMENT "Generating menu.json from .env / defaults..."
            VERBATIM
        )
    else()
        add_custom_command(
            OUTPUT  ${MENU_OUT}
            COMMAND ${CMAKE_COMMAND} -E echo "menu.json already exists, skipping"
            COMMENT "menu.json pre-generated, skipping"
        )
    endif()

    add_custom_target(generate_menu DEPENDS ${MENU_OUT})
endif()

# ─── Material Icons font generation ──────────────────────────────────────────
# Generates material_icons_used_24.c from buttons.json icon references.
# If the .c file already exists (pre-generated by CI step), skip generation.
set(GEN_ICONS_SCRIPT "${COMPONENT_SRCDIR}/scripts/gen_icons.py")
set(MI_TTF           "${COMPONENT_SRCDIR}/scripts/MaterialIcons-Regular.ttf")
set(MI_CODEPOINTS    "${COMPONENT_SRCDIR}/scripts/MaterialIcons-Regular.codepoints")
set(BUTTONS_JSON     "${COMPONENT_SRCDIR}/buttons.json")
set(MI_FONT_C        "${COMPONENT_SRCDIR}/material_icons_used_24.c")

if(NOT CMAKE_SCRIPT_MODE_FILE)
    if(NOT EXISTS "${MI_FONT_C}")
        add_custom_command(
            OUTPUT  ${MI_FONT_C}
            COMMAND ${Python3_EXECUTABLE} ${GEN_ICONS_SCRIPT}
            DEPENDS ${GEN_ICONS_SCRIPT} ${MI_TTF} ${MI_CODEPOINTS}
                    ${BUTTONS_JSON} ${MENU_OUT}
            WORKING_DIRECTORY ${COMPONENT_SRCDIR}/scripts
            COMMENT "Generating Material Icons subset font..."
            VERBATIM
        )
    else()
        add_custom_command(
            OUTPUT  ${MI_FONT_C}
            COMMAND ${CMAKE_COMMAND} -E echo "material_icons_used_24.c pre-generated, skipping"
            COMMENT "Material Icons font already exists, skipping generation"
        )
    endif()

    add_custom_target(generate_mi_font DEPENDS ${MI_FONT_C})
endif()

# ─── Component registration ───────────────────────────────────────────────────
idf_component_register(
    SRCS
        "main.cpp"
        "init.cpp"
        "ble_config.cpp"
        "bluetoothFunctions.cpp"
        "setupFunctions.cpp"
        "fileFunctions.cpp"
        "menuFunctions.cpp"
        "functions.cpp"
        "mcs_client.cpp"
        "iconManager.cpp"
        "material_icons_used_24.c"
        "buttonUI.cpp"
        "webServer.cpp"
        "wifiManager.cpp"
        ${UI_C_SOURCES}
        ${UI_CPP_SOURCES}
    INCLUDE_DIRS
        "."
        "ui"
    EMBED_TXTFILES
        "buttons.json"
        "menu.json"
    EMBED_FILES
        ${EMBEDDED_IMAGES}
    REQUIRES
        json
        joltwallet__littlefs
        bt
        nvs_flash
        led_strip
        esp32-bq27441
        esp32-TCA8418
        TFT_eSPI
        esp_adc
        espressif__arduino-esp32
        lvgl__lvgl
        espressif__esp_lvgl_port
        esp_wifi
        esp_netif
        esp_event
        esp_http_server
        esp32async__asynctcp
        esp32async__espasyncwebserver
)

# ─── Build order dependencies ─────────────────────────────────────────────────
# menu.json must exist before font generation (gen_icons.py scans it for icons)
# Font must exist before compiling material_icons_used_24.c
if(NOT CMAKE_SCRIPT_MODE_FILE)
    add_dependencies(generate_mi_font generate_menu)
    add_dependencies(__idf_main generate_mi_font)
    add_dependencies(__idf_main generate_menu)
endif()

# ─── LittleFS partition image ─────────────────────────────────────────────────
littlefs_create_partition_image(
    storage
    data
    FLASH_IN_PROJECT
)

# ─── Compiler settings ───────────────────────────────────────────────────────
target_compile_definitions(${COMPONENT_LIB} PUBLIC
    LV_CONF_PATH="${COMPONENT_SRCDIR}/lv_conf.h"
)

idf_component_get_property(lvgl_lib lvgl__lvgl COMPONENT_LIB)
target_compile_definitions(${lvgl_lib} PUBLIC
    LV_CONF_PATH="${COMPONENT_SRCDIR}/lv_conf.h"
)

target_compile_options(${COMPONENT_LIB} PRIVATE
    -Wno-unused-const-variable
    -Wno-missing-field-initializers
)

set(CMAKE_C_FLAGS   "${CMAKE_C_FLAGS}   -Wno-error -Wno-format")
set(CMAKE_CXX_FLAGS "${CMAKE_CXX_FLAGS} -Wno-error -Wno-format")
